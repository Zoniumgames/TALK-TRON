cmake_minimum_required(VERSION 3.22)
project(TalkTron VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(TalkTron COMPANY_NAME "Zonium" BUNDLE_ID "com.zonium.talktron" PLUGIN_MANUFACTURER_CODE Zoni PLUGIN_CODE TlTr FORMATS VST3 Standalone PRODUCT_NAME "TALK-TRON" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(TalkTron)
target_sources(TalkTron PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(TalkTron PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TalkTron PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
